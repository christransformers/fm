erHooks() {
    const g = typeof window !== 'undefined' ? window : globalThis;
    if (typeof g.remember === 'function' && !g.remember.__fuseFreqHooked) {
      const originalRemember = g.remember;
      function rememberHooked(id) {
        originalRemember(id);
        try { recordPlay(id); } catch {}
      }
      rememberHooked.__fuseFreqHooked = true;
      g.remember = rememberHooked;
    }
    if (typeof g.nextIndexForPlayback === 'function' && !g.nextIndexForPlayback.__fuseFreqHooked) {
      const originalNext = g.nextIndexForPlayback;
      function nextHooked() {
        try {
          if (freq.nightDrive) {
            const fi = nightDriveNextIndex();
            if (Number.isInteger(fi) && fi >= 0) return fi;
          }
        } catch {}
        return originalNext();
      }
      nextHooked.__fuseFreqHooked = true;
      g.nextIndexForPlayback = nextHooked;
    }
    // Clear frequency map with full reset
    const clearAll = document.getElementById('clearAllState');
    if (clearAll && !clearAll.__fuseFreqHooked) {
      clearAll.__fuseFreqHooked = true;
      clearAll.addEventListener('click', () => {
        try { localStorage.removeItem(FREQ_KEY); } catch {}
      }, true);
    }
  }

  function mount() {
    installPlayerHooks();
    refreshUI();
    animateRadar();
    document.addEventListener('click', onAction);
    document.addEventListener('keydown', onKey);

    // Sync initial state classes
    if (freq.nightDrive) {
      document.body.classList.add('night-drive-on');
      const overlay = document.getElementById('nightDriveOverlay');
      if (overlay) overlay.hidden = false;
    }
    // Re-hook after a tick in case app.js finishes later (same defer order is fine)
    setTimeout(installPlayerHooks, 0);
  }

  window.fuseFrequency = {
    onPlay: recordPlay,
    nightDriveNextIndex,
    isNightDrive: () => !!freq.nightDrive,
    toggleNightDrive,
    togglePanel,
    downloadCard,
    refresh: refreshUI
  };
  // Soft-refresh overlay when the now-playing title changes
  const titleEl = document.getElementById('nowTitle');
  if (titleEl && typeof MutationObserver !== 'undefined') {
    new MutationObserver(() => refreshUI()).observe(titleEl, { childList: true, characterData: true, subtree: true });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', mount);
  } else {
    // app.js is defer; this script is also defer — run after both parse
    mount();
  }
})();
